	\begin{lecture}{中国剩余定理}
		\textbf{定理 CRT.} 设 $m_1,m_2,\ldots,m_n$ 两两互质，且
		\[
		(S):
		\left\{
		\begin{matrix}
			x \equiv a_1 \pmod{m_1},\\
			x \equiv a_2 \pmod{m_2},\\
			\vdots\\
			x \equiv a_n \pmod{m_n}.
		\end{matrix}
		\right.
		\]
		令
		\[
			M=\prod_{i=1}^{n}m_i,\qquad
			M_i=\frac{M}{m_i},\qquad
			t_i\equiv M_i^{-1}\pmod{m_i}.
		\]
		则方程组 $(S)$ 有解，并且一组解为
		\[
			x\equiv \sum_{i=1}^{n} a_i t_i M_i \pmod M.
		\]
		\textbf{构造性证明.} 由于 $m_1,m_2,\ldots,m_n$ 两两互质，对任意 $i$，有
		\[
			\gcd(M_i,m_i)=1.
		\]
		因此存在 $t_i$，使得
		\[
			t_iM_i\equiv 1\pmod{m_i}.
		\]
		令
		\[
			x=\sum_{k=1}^{n}a_k t_k M_k.
		\]
		固定任意 $i$。当 $k\ne i$ 时，由于 $m_i\mid M_k$，有
		\[
			a_k t_k M_k\equiv 0\pmod{m_i}.
		\]
		于是
		\[
		\begin{aligned}
			x
			&=a_i t_i M_i+\sum_{k\ne i}a_k t_k M_k\\
			&\equiv a_i\cdot 1+\sum_{k\ne i}0 \pmod{m_i}\\
			&\equiv a_i \pmod{m_i}.
		\end{aligned}
		\]
		因此 $x$ 同时满足所有同余式，故 $x$ 是方程组 $(S)$ 的一组解。
		
	\end{lecture}

	\begin{lecture}{模逆元}
		\[
		\begin{aligned}
			& \dfrac{a}{b} \equiv a \cdot b_{\mathrm{inv}} \equiv x \pmod p\\
			\Rightarrow & a \equiv bx \pmod p\\
			\Rightarrow & a \cdot b_{\mathrm{inv}} \cdot b \equiv bx \pmod p\\
			\Rightarrow & a \cdot (b_{\mathrm{inv}} \cdot b - 1) \equiv 0 \pmod p\\
			\Rightarrow & b_{\mathrm{inv}} \cdot b \equiv 1 \pmod p
		\end{aligned}
		\]
		\[
		\begin{aligned}
			\mathrm{inv} \text{存在} \Leftrightarrow & \gcd(b,p) = 1\\
			b \cdot b_{\mathrm{inv}} = k \cdot p + 1,\ g = \gcd(b,p) 
			\Rightarrow & 
			g \cdot b_1 \cdot b_{\mathrm{inv}} 
			= k \cdot g \cdot p_1 + 1\\
			\Rightarrow & 
			g \cdot (b_1 \cdot b_{\mathrm{inv}} - k \cdot p_1) = 1\\
			\text{数域为} \mathbb{Z} 
			\Rightarrow & 
			g = 1,\ b_1 \cdot b_{\mathrm{inv}} - k \cdot p_1 = 1
		\end{aligned}
		\]
	\end{lecture}

	\begin{lecture}{线性求逆元}
		$p$ 为质数，$p > n$
		\[
		\begin{array}{c}
		\forall x \in \mathbb{N}^+,\quad
		p = kx + r,\quad
		k = \left\lfloor \frac{p}{x} \right\rfloor,\quad
		r = p \bmod x
		\\[6pt]
		\begin{array}{rl}
		kx + r \equiv 0 \pmod p
		\Rightarrow & kx \equiv -r \pmod p
		\\
		\Rightarrow & k \cdot r_{\mathrm{inv}} \cdot x \equiv -1 \pmod p
		\\
		x \cdot x_{\mathrm{inv}} \equiv 1 \pmod p
		\Rightarrow & k \cdot r_{\mathrm{inv}} \cdot x \equiv -x \cdot x_{\mathrm{inv}} \pmod p
		\\
		\Rightarrow & x_{\mathrm{inv}} \equiv -k \cdot r_{\mathrm{inv}} \pmod p
		\end{array}
		\\[6pt]
		r = p \bmod x < x \Rightarrow r_{\mathrm{inv}} \text{ 比 } x_{\mathrm{inv}} \text{ 先被求出}
		\end{array}
		\]
	\end{lecture}
