\section{方程}
\[ax^2+bx+c=0 \Rightarrow x = \frac{-b\pm\sqrt{b^2-4ac}}{2a}\]

极值点 $x = -\frac{b}{2a}$.

\[\begin{aligned}ax+by=e\\cx+dy=f\end{aligned}
\Rightarrow
\begin{aligned}x=\dfrac{ed-bf}{ad-bc}\\y=\dfrac{af-ec}{ad-bc}\end{aligned}\]

一般地，给定方程 $Ax = b$, 变量 $x_i$ 的解为
\[x_i = \frac{\det A_i'}{\det A} \]
其中 $A_i'$ 表示将 $A$ 的第 $i$ 列替换为 $b$.

\sin(v+w)&{}=\sin v\cos w+\cos v\sin w\\
\cos(v+w)&{}=\cos v\cos w-\sin v\sin w\\
\end{align*}
\begin{align*}
\tan(v+w)&{}=\dfrac{\tan v+\tan w}{1-\tan v\tan w}\\
\sin v+\sin w&{}=2\sin\dfrac{v+w}{2}\cos\dfrac{v-w}{2}\\
\cos v+\cos w&{}=2\cos\dfrac{v+w}{2}\cos\dfrac{v-w}{2}
\end{align*}
\[ (V+W)\tan(v-w)/2{}=(V-W)\tan(v+w)/2 \]
其中 $V, W$ 是角 $v, w$ 所对边的长度.
\begin{align*}
	a\cos x+b\sin x&=r\cos(x-\phi)\\
	a\sin x+b\cos x&=r\sin(x+\phi)
\end{align*}
其中 $r=\sqrt{a^2+b^2}, \phi=\operatorname{atan2}(b,a)$.

\subsection{三角形}
边长: $a,b,c$\\
半周长: $p=\dfrac{a+b+c}{2}$\\
面积: $A=\sqrt{p(p-a)(p-b)(p-c)}$\\
外接圆半径: $R=\dfrac{abc}{4A}$\\
内切圆半径: $r=\dfrac{A}{p}$\\
中线长度 (将三角形分为两个面积相等的三角形): $m_a=\tfrac{1}{2}\sqrt{2b^2+2c^2-a^2}$\\
角平分线长度 (将角度分为两个相等的角度): $s_a=\sqrt{bc\left[1-\left(\dfrac{a}{b+c}\right)^2\right]}$\\
正弦定理: $\dfrac{\sin\alpha}{a}=\dfrac{\sin\beta}{b}=\dfrac{\sin\gamma}{c}=\dfrac{1}{2R}$\\
余弦定理: $a^2=b^2+c^2-2bc\cos\alpha$\\
正切定理: $\dfrac{a+b}{a-b}=\dfrac{\tan\dfrac{\alpha+\beta}{2}}{\tan\dfrac{\alpha-\beta}{2}}$\\

\subsection{四边形}
设边长为 $a,b,c,d$, 半周长 $p$, 对角线为 $e, f$, 对角线夹角 $\theta$, 面积 $A$ 和
辅助量 $F=b^2+d^2-a^2-c^2$:

\[ 4A = 2ef \cdot \sin\theta = F\tan\theta = \sqrt{4e^2f^2-F^2} \]

对于圆内接四边形，对角之和为 $180^\circ$,
$ef = ac + bd$, 且 $A = \sqrt{(p-a)(p-b)(p-c)(p-d)}$.

\section{导数/积分}
\begin{align*}
	\dfrac{d}{dx}\arcsin x = \dfrac{1}{\sqrt{1-x^2}} &&& \dfrac{d}{dx}\arccos x = -\dfrac{1}{\sqrt{1-x^2}} \\
	\dfrac{d}{dx}\tan x = 1+\tan^2 x &&& \dfrac{d}{dx}\arctan x = \dfrac{1}{1+x^2} \\
	\int\tan ax = -\dfrac{\ln|\cos ax|}{a} &&& \int x\sin ax = \dfrac{\sin ax-ax \cos ax}{a^2} \\
	\int e^{-x^2} = \frac{\sqrt \pi}{2} \text{erf}(x) &&& \int xe^{ax}dx = \frac{e^{ax}}{a^2}(ax-1)
\end{align*}

分部积分:
\[\int_a^bf(x)g(x)dx = [F(x)g(x)]_a^b-\int_a^bF(x)g'(x)dx\]

\section{概率论}
设 $X$ 离散随机变量，
取值为 $x$ 的概率为 $p_X(x)$。
期望值(均值)为 $\mu=\mathbb{E}(X)=\sum_xxp_X(x)$ ，
方差为 $\sigma^2=V(X)=\mathbb{E}(X^2)-(\mathbb{E}(X))^2=\sum_x(x-\mathbb{E}(X))^2p_X(x)$ 
其中 $\sigma$ 是标准差。
如果 $X$ 是连续随机变量，它将具有概率密度函数 $f_X(x)$，上述求和将变为积分，$p_X(x)$ 被 $f_X(x)$ 替换。

期望值具有线性性：
\[\mathbb{E}(aX+bY) = a\mathbb{E}(X)+b\mathbb{E}(Y)\]
对于相互独立的 $X$ 和 $Y$, \[V(aX+bY) = a^2V(X)+b^2V(Y).\]

\subsection{离散分布}

\subsubsection{二项分布}
在 $n$ 次相互独立的是/否实验中，
每次实验以概率 $p$ 成功，
则成功次数服从 $\textrm{Bin}(n,p),\,n=1,2,\dots,\, 0\leq p\leq1$.
\[p(k)=\binom{n}{k}p^k(1-p)^{n-k}\]
\[\mu = np,\,\sigma^2=np(1-p)\]
当 $p$ 较小时，$\textrm{Bin}(n,p)$ 可近似为 $\textrm{Po}(np)$。

\subsubsection{首次成功分布}
在 $n$ 次相互独立的是/否实验中，
每次实验以概率 $p$ 成功，
则首次成功所需的试验次数服从 $\textrm{Fs}(p),\,0\leq p\leq1$.
\[p(k)=p(1-p)^{k-1},\,k=1,2,\dots\]
\[\mu = \frac1p,\,\sigma^2=\frac{1-p}{p^2}\]

\subsubsection{泊松分布}
如果事件以已知平均速率 $\kappa$ 发生，
并且与距离上一次事件发生的时间相互独立，
那么固定时间段 $t$ 内发生的事件数服从 $\textrm{Po}(\lambda),\,\lambda=t\kappa$.
\[p(k)=e^{-\lambda}\frac{\lambda^k}{k!}, k=0,1,2,\dots\]
\[\mu=\lambda,\,\sigma^2=\lambda\]
